@router.post("/train-categories")
def train_categories() -> Dict[str, Any]:
    """
    Train / retrain the ML-based category classifier
    from existing labelled bill items.
    """
    db = SessionLocal()
    try:
        n_samples = train_category_model(db)
        if n_samples == 0:
            return {
                "message": "Not enough data or scikit-learn not installed. Using keyword-based categories.",
                "samples_used": 0,
            }

        return {
            "message": "Category model trained successfully.",
            "samples_used": n_samples,
        }
    finally:
        db.close()
